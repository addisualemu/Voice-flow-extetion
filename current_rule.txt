rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    // Users collection
    match /users/{userId} {
      // Users can read and write their own user document
      allow read, write: if request.auth != null && request.auth.uid == userId;

      // Tasks subcollection
      match /tasks/{taskId} {
        // Users can only access their own tasks
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }

    // Pins collection - for PIN to userId mapping
    // Document ID is the PIN, document contains { userId: string }
    match /linkingPins/{pin} {
      // Anyone authenticated can read PINs (for uniqueness checking and external app lookups)
      allow read: if request.auth != null;
      
      // Users can create/update PIN entries that map to their own userId
      allow create, update: if request.auth != null && 
        request.resource.data.userId == request.auth.uid;
      
      // Users can delete PIN entries that map to their own userId
      allow delete: if request.auth != null && 
        resource.data.userId == request.auth.uid;
    }
  }
}
